# Main application environment — the LangGraph copilot, MCP client/server,
# RAG, memory, guardrails, and agent evaluation.
#
# Install with:
#   python -m venv .venv
#   source .venv/bin/activate
#   python -m pip install -r requirements.txt
#
# NOTE on the two-venv setup (see README "Why two virtual environments"):
# arize-phoenix's full package transitively requires mcp>=2.0 (via
# pydantic-ai -> fastmcp), while our own MCP server/client stack needs
# mcp<2 (the stable FastMCP v1 API + a compatible langchain-mcp-adapters
# release). These cannot coexist in one environment. This venv sends
# real OpenTelemetry/OpenInference spans over the wire (no `phoenix`
# package import needed for that) to a Phoenix server run from the
# separate environment in requirements-phoenix.txt.
